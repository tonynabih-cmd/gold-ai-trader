### risk.js

**Purpose:** 18-rule safety gate. Every rule checked before ANY trade.

```javascript
// 1. Kill switch
if (process.env.BOT_ENABLED !== 'true')
  return 'SKIP: Bot disabled';

// 2. Golden Hour - 12PM-8PM UAE = 08:00-16:00 UTC
const hour = now.getUTCHours();
if (hour < 8 || hour >= 16)
  return 'SKIP: Outside Golden Hour (UAE 12PM-8PM)';

// 3. Friday close at 8PM UAE
const day = now.getUTCDay();
if (day === 5 && hour >= 16)
  return 'SKIP: Friday close - weekend approaching';

// 4. Weekend
if (day === 6 || day === 0)
  return 'SKIP: Weekend';

// 5. Spread from OANDA bid/ask (NOT Twelve Data)
const spread = ask - bid;
if (spread > 0.40)
  return 'SKIP: Spread too high - possible news event';

// 6. ATR range
if (atr < 1 || atr > 8)
  return 'SKIP: ATR out of range';

// 7. Daily trade cap
if (botState.dailyTrades >= 5)
  return 'SKIP: Daily trade limit (5/5)';

// 8. Daily loss limit - 5% of account ($5 on $100)
if (parseFloat(botState.dailyLoss) >= parseFloat(botState.balance) * 0.05)
  return 'STOP: Daily loss limit reached';

// 9. Total drawdown - 20% shutdown
if (parseFloat(botState.totalDrawdown) >= 20)
  return 'DISABLE: Max drawdown - bot disabled permanently';

// 10. Cooldown - 10 minutes between trades
if (minutesSinceLastTrade < 10)
  return 'SKIP: Cooldown active';

// 11. Max open positions
if (botState.openTrades.length >= 2)
  return 'SKIP: Max 2 positions open';

// 12. Duplicate trade ID
if (botState.recentTradeIds.includes(signal.id))
  return 'SKIP: Duplicate order';

// 13. Slippage protection
if (Math.abs(currentPrice - signal.entryPrice) > 0.30)
  return 'SKIP: Price moved too fast';

// 14. Sufficient balance
if (parseFloat(botState.balance) < minimumRequired)
  return 'SKIP: Insufficient balance';

// 15. Duplicate candle
if (lastCandleTime <= previousProcessedCandleTime)
  return 'SKIP: Duplicate candle';

// 16. Score check
if (signal.score < 2)
  return 'SKIP: Score too low';

// 17. ATR spike
if (atr > atrAverage * 2)
  return 'SKIP: Abnormal volatility spike';

// 18. Order timestamp guard - retry protection
if (Date.now() - lastOrderTimestamp < 60000)
  return 'SKIP: Too soon after last order';

return 'APPROVED';
```

### execution.js

**Purpose:** Place real orders on OANDA. Only runs after risk.js returns APPROVED.

```javascript
async function placeTrade(signal, botState) {

  // 1. Save backup state first
  await saveBackupState(botState);

  // 2. Store trade ID - keep only last 20
  botState.recentTradeIds.push(signal.id);
  botState.recentTradeIds = botState.recentTradeIds.slice(-20);

  // 3. Stop loss - correct direction
  const stopLoss = signal.action === 'BUY'
    ? signal.entryPrice - (1.5 * signal.atr)  // BUY: below entry
    : signal.entryPrice + (1.5 * signal.atr); // SELL: above entry

  // 4. Dynamic position sizing
  const riskAmount = parseFloat(botState.balance) * 0.01; // 1% = $1
  const stopDistance = Math.abs(signal.entryPrice - stopLoss);
  const positionSize = riskAmount / stopDistance;

  // 5. Build OANDA order - all prices as strings 3 decimals
  const orderBody = {
    order: {
      type: 'MARKET',
      instrument: 'XAU_USD',
      units: signal.action === 'BUY'
        ? positionSize.toFixed(3)
        : (-positionSize).toFixed(3),
      timeInForce: 'FOK', // Fill or Kill - no partial fills
      stopLossOnFill: {
        price: stopLoss.toFixed(3),
        timeInForce: 'GTC' // Stays active even if bot crashes
      },
      takeProfitOnFill: {
        price: signal.takeProfit.toFixed(3),
        timeInForce: 'GTC'
      }
    }
  };

  // 6. Send to OANDA (practice or live based on env)
  const baseUrl = process.env.OANDA_ENV === 'practice'
    ? 'https://api-fxpractice.oanda.com'
    : 'https://api-fxtrade.oanda.com';

  const result = await fetch(
    `${baseUrl}/v3/accounts/${process.env.OANDA_ACCOUNT_ID}/orders`,
    {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${process.env.OANDA_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(orderBody)
    }
  ).then(r => r.json());

  // 7. Handle FOK rejection gracefully
  if (!result.orderFillTransaction) {
    await logger.save({
      ...signal,
      tradeExecuted: false,
      reason: 'REJECTED_BY_MARKET - price moved, waiting next cycle'
    });
    return 'FOK_REJECTED';
  }

  // 8. Update state with actual filled data
  const filled = result.orderFillTransaction;
  botState.openTrades.push({
    tradeId: signal.id,
    pair: 'XAU_USD',
    entry: parseFloat(filled.price),
    units: parseFloat(filled.units),
    stopLoss: stopLoss,
    takeProfit: signal.takeProfit,
    openedAt: Date.now(),
    strategyVersion: 'v1.0'
  });

  // 9. Update counters
  botState.dailyTrades++;
  await state.save(botState);

  // 10. Send email alert
  await sendAlert(`✅ ${signal.action} ${positionSize.toFixed(3)}oz XAU/USD at $${filled.price}`);
}
```

### cron.js

**Purpose:** Main pipeline. Runs every 5 min via UptimeRobot.

```javascript
export default async function handler(req, res) {

  // Verify authorization
  if (req.headers['authorization'] !== `Bearer ${process.env.CRON_SECRET}`) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  try {
    // Pipeline
    const botState = await startup();
    dailyReset(botState);

    const candles = await marketData.fetchCandles(botState);
    if (candles.skip) return res.json({ skipped: candles.reason });

    const indicators = await indicators.calculate(candles);
    if (indicators.skip) return res.json({ skipped: indicators.reason });

    const signal = await strategy.generateSignal(indicators, botState);
    if (signal) await saveSignalToQueue(signal);

    const riskResult = await risk.checkAllRules(signal, botState, indicators);
    if (riskResult !== 'APPROVED') {
      await logger.save({ ...signal, tradeExecuted: false, reason: riskResult });
      return res.json({ skipped: riskResult });
    }

    await execution.placeTrade(signal, botState);
    await logger.save({ ...signal, tradeExecuted: true });
    await state.save(botState);
    await monitor.heartbeat(botState);

    return res.json({ success: true, decision: signal.action });

  } catch (err) {
    await sendAlert(`🚨 Bot error: ${err.message}`);
    return res.status(500).json({ error: err.message });
  }
}
```
